Code Block 1: Generic example of model construction using A) S3 objects (mlr) and B) R6 objects (mlr3). In both approaches we are randomly drawing a training (train.set) and test (test.set) dataset from our full dataset (full.dataset). The model stacking ensemble that we use in B requires mlr3pipelines.

ensemble.model <- function(df.train, df.test){
  
  df.prdctr <- df.train %>% select(predictor) %>% distinct() %>% pull()
  df.trn <- df.train %>% select(-predictor)
  df.tst <- df.test %>% select(-predictor)
  
  sample.tsk <- as_task_classif(x=df.trn, target="mpb")
  sample.tst <- as_task_classif(x=df.tst, target="mpb")

  m.e.cv = GraphLearner$new(m.1, predict_type="prob")  
  m.train <- m.e.cv$train(sample.tsk)
  m.prd <- m.e.cv$predict(sample.tst)
  
  m.prf <- data.frame(
    acc = m.prd$score(msr("classif.acc")),
    auc = m.prd$score(msr("classif.auc")),
    ce = m.prd$score(msr("classif.ce")),
    fdr = m.prd$score(msr("classif.fdr")),
    fnr = m.prd$score(msr("classif.fnr")),
    fomr = m.prd$score(msr("classif.fomr")),
    fpr = m.prd$score(msr("classif.fpr")),
    tnr = m.prd$score(msr("classif.tnr")),
    tpr = m.prd$score(msr("classif.tpr"))) %>%
    as_tibble() %>% bind_cols(t(m.train$timings) %>% as_tibble()) %>% 
    mutate(predictor=df.prdctr)

  return(m.prf)
  
}

var.selex <- function(train.df, test.df, max.feat=14, cutoff=0.8, iter=j){
  
  #### correlation matrix ####
  # NOTE: this is messy and I'll rewrite it at some point
  
  corr.mat <- bind_rows(
    train.df %>% 
    bind_rows(test.df) %>%
    select(-mpb) %>% 
    as.matrix() %>% 
    cor(., method="pearson") %>%
    as.data.frame() %>%
    rownames_to_column() %>%
    pivot_longer(-rowname) %>%
    mutate(method="pearson"),
    train.df %>% 
    bind_rows(test.df) %>%
    select(-mpb) %>% 
    as.matrix() %>% 
    cor(., method="spearman") %>%
    as.data.frame() %>%
    rownames_to_column() %>%
    pivot_longer(-rowname) %>%
    mutate(method="spearman")) %>%
    mutate(abs.value=abs(value)) %>%
    select(var.1=rowname,
           var.2=name,
           method, abs.value) %>%
    pivot_wider(values_from=abs.value, names_from=method)
  
  #### result.grid  ####
  
  result.grid <- list()

  for(i in 1:max.feat){
    
  print(paste0("#### model ", i, " ####"))  
    
  if(i == 1){  

  train.split <- train.df %>%
  pivot_longer(-mpb, names_to="predictor", values_to="obs") %>%
  split(.$predictor, drop=TRUE) 
  
  test.split <- test.df %>%
  pivot_longer(-mpb, names_to="predictor", values_to="obs") %>%
  split(.$predictor)    
  
  model <- map2(.x=train.split, .y=test.split, .f=ensemble.model) %>% bind_rows()  
  
result.grid[[i]] <- p.s <- model %>% mutate(n.feat=i)
   
  }else{
    
  p.s <- result.grid %>% bind_rows() %>% group_by(n.feat) %>%
    slice_min(order_by=ce) %>% ungroup()
  
  crm <- corr.mat %>%
    filter(var.1 %in% p.s$predictor) %>%
    filter(pearson >= cutoff | spearman >= cutoff)
  
  can.var <- corr.mat %>% filter(var.2 %ni% crm$var.2) %>%
    select(var.2) %>% distinct()
    
  if(nrow(can.var) > 0){
  
  train.split <- train.df %>%
    select(mpb, contains(p.s$predictor), contains(can.var$var.2)) %>%
    pivot_longer(!c(mpb, contains(p.s$predictor)), names_to="predictor", values_to="obs") %>%
    split(.$predictor, drop=TRUE) 
  
  test.split <- test.df %>%
    select(mpb, contains(p.s$predictor), contains(can.var$var.2)) %>%
    pivot_longer(!c(mpb, contains(p.s$predictor)), names_to="predictor", values_to="obs") %>%
    split(.$predictor, drop=TRUE) 
  
  model <- map2(.x=train.split, .y=test.split, .f=ensemble.model) %>% bind_rows()
  
  result.grid[[i]] <- p.s <- model %>% mutate(n.feat=i)
  }
  
  print("done")
  
  }  
    
    
  }
  
  results <- result.grid %>% bind_rows() %>%
    write_csv(paste0("var.selex/vs.", iter, ".csv"))
  
}
